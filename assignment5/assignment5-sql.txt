SELECT line_items.order_id, SUM(products.price * line_items.quantity) AS total_price FROM line_items JOIN orders ON line_items.order_id = orders.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY line_items.order_id ORDER BY line_items.order_id LIMIT 5; 


SELECT c.customer_name, AVG(t.total_price) AS average_order_price FROM customers c JOIN (SELECT line_items.order_id, orders.customer_id, SUM(products.price * line_items.quantity) AS total_price FROM line_items JOIN orders ON line_items.order_id = orders.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY line_items.order_id, orders.customer_id) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;
